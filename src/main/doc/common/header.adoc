<script type="module">
  let pagefind;
  async function initPagefind() {
    if (!pagefind) {
      pagefind = await import('/pagefind/pagefind.js');
      await pagefind.init();
    }
    return pagefind;
  }

  function render(results, query) {
    const box = document.getElementById("search-results");
    if (!results.length) {
      box.innerHTML = `<div class="no-results">No results for "<strong>${query}</strong>"</div>`;
    } else {
      box.innerHTML = results.map((r) =>
        `<a href="${r.url.replace(/\.html$/, '')}">
          <div class="result-title">${r.meta.title || r.url}</div>
          <div class="result-summary">${r.excerpt}</div>
        </a>`
      ).join("");
    }
    box.classList.add("visible");
  }

  let debounce = null;

  document.addEventListener("DOMContentLoaded", function() {
    const input = document.getElementById("blog-search");
    const box   = document.getElementById("search-results");

    input.addEventListener("input", () => {
      clearTimeout(debounce);
      const q = input.value.trim();
      if (!q) { box.classList.remove("visible"); return; }
      debounce = setTimeout(async () => {
        try {
          const pf = await initPagefind();
          const search = await pf.search(q);
          const results = await Promise.all(search.results.slice(0, 8).map(r => r.data()));
          render(results, q);
        } catch (e) {
          console.error("Search failed:", e);
        }
      }, 250);
    });

    document.addEventListener("click", (e) => {
      if (!e.target.closest(".search-wrapper")) {
        box.classList.remove("visible");
      }
    });
  });
</script>
