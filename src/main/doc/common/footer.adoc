<script type="module">
  const REACTIONS_API = window.location.hostname === "localhost"
    ? "http://localhost:4321/api/reactions"
    : "https://anton-blog-0ac9306d-antonkol.wix-site-host.com/api/reactions";
  const COMMENTS_API = window.location.hostname === "localhost"
    ? "http://localhost:4321/api/comments"
    : "https://anton-blog-0ac9306d-antonkol.wix-site-host.com/api/comments";

  const NON_ARTICLE = ["/home", "/about"];
  const path = window.location.pathname;
  const isArticle = !NON_ARTICLE.some((p) => path === p || path.endsWith(p))
                 && !!document.querySelector(".sect1 > h2");

  /* --- Reactions --- */

  function reactionsKey(postUrl) { return `reactions:${postUrl}`; }

  function getVoted(postUrl) {
    try { return new Set(JSON.parse(localStorage.getItem(reactionsKey(postUrl)) || "[]")); }
    catch { return new Set(); }
  }

  function saveVoted(postUrl, type) {
    const voted = getVoted(postUrl);
    voted.add(type);
    localStorage.setItem(reactionsKey(postUrl), JSON.stringify([...voted]));
  }

  function renderReactions(counts, voted, postUrl) {
    const section = document.getElementById("reactions-section");
    section.innerHTML = [
      { type: "like", emoji: "👍" },
      { type: "heart", emoji: "❤️" },
    ].map(({ type, emoji }) => `
      <button class="reaction-btn${voted.has(type) ? " active" : ""}" data-type="${type}">
        ${emoji} <span>${counts[type] ?? 0}</span>
      </button>`).join("");

    section.querySelectorAll(".reaction-btn").forEach((btn) => {
      btn.addEventListener("click", async () => {
        const type = btn.dataset.type;
        if (voted.has(type)) return;
        btn.disabled = true;
        try {
          const res = await fetch(REACTIONS_API, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ url: postUrl, type }),
          });
          if (!res.ok) return;
          const newCounts = await res.json();
          saveVoted(postUrl, type);
          renderReactions(newCounts, getVoted(postUrl), postUrl);
        } catch {
          btn.disabled = false;
        }
      });
    });
  }

  /* --- Comments --- */

  function formatDate(iso) {
    return new Date(iso).toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" });
  }

  function renderComments(comments) {
    const list = document.getElementById("comments-list");
    if (!comments.length) {
      list.innerHTML = '<p class="comment-status">No comments yet. Be the first!</p>';
      return;
    }
    list.innerHTML = comments.map((c) => `
      <div class="comment">
        <div class="comment-meta">
          <span class="comment-author">${c.author}</span>
          <span class="comment-date">${formatDate(c.createdAt)}</span>
        </div>
        <div class="comment-content">${c.content}</div>
      </div>`).join("");
  }

  async function loadComments() {
    const postUrl = window.location.origin + window.location.pathname;
    const list = document.getElementById("comments-list");
    list.innerHTML = '<p class="comment-status">Loading comments…</p>';
    try {
      const res = await fetch(`${COMMENTS_API}?url=${encodeURIComponent(postUrl)}`);
      renderComments(res.ok ? await res.json() : []);
    } catch {
      list.innerHTML = '<p class="comment-status">Could not load comments.</p>';
    }
  }

  /* --- Init --- */

  document.addEventListener("DOMContentLoaded", async () => {
    if (!isArticle) return;

    const postUrl = window.location.origin + window.location.pathname;

    const reactionsSection = document.getElementById("reactions-section");
    reactionsSection.style.display = "";
    try {
      const res = await fetch(`${REACTIONS_API}?url=${encodeURIComponent(postUrl)}`);
      const counts = res.ok ? await res.json() : { like: 0, heart: 0 };
      renderReactions(counts, getVoted(postUrl), postUrl);
    } catch {
      reactionsSection.style.display = "none";
    }

    document.querySelector(".comments-section").style.display = "";
    loadComments();

    document.getElementById("comment-submit").addEventListener("click", async () => {
      const author  = document.getElementById("comment-author").value.trim();
      const content = document.getElementById("comment-content").value.trim();
      const status  = document.getElementById("comment-status");
      const btn     = document.getElementById("comment-submit");

      if (!author || !content) { status.textContent = "Please fill in both fields."; return; }

      btn.disabled = true;
      status.textContent = "Posting…";

      try {
        const res = await fetch(COMMENTS_API, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ url: postUrl, author, content }),
        });
        if (!res.ok) throw new Error();
        document.getElementById("comment-author").value = "";
        document.getElementById("comment-content").value = "";
        status.textContent = "Comment posted!";
        await loadComments();
      } catch {
        status.textContent = "Failed to post comment. Please try again.";
      } finally {
        btn.disabled = false;
      }
    });
  });
</script>
<script type="module">
  const SUBSCRIBE_API = window.location.hostname === "localhost"
    ? "http://localhost:4321/api/subscribe"
    : "https://anton-blog-0ac9306d-antonkol.wix-site-host.com/api/subscribe";

  document.addEventListener("DOMContentLoaded", () => {
    const btn = document.getElementById("sub-btn");
    if (!btn) return;
    btn.addEventListener("click", async () => {
      const email  = document.getElementById("sub-email").value.trim();
      const status = document.getElementById("sub-status");
      const btn    = document.getElementById("sub-btn");

      if (!email) { status.textContent = "Please enter your email."; return; }

      btn.disabled = true;
      status.textContent = "Subscribing…";

      try {
        const res = await fetch(SUBSCRIBE_API, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ email }),
        });
        const data = await res.json();
        if (!res.ok) { status.textContent = data.error ?? "Something went wrong."; return; }
        status.textContent = "Subscribed! Thanks for following along.";
        document.getElementById("sub-email").value = "";
      } catch {
        status.textContent = "Network error, please try again.";
      } finally {
        btn.disabled = false;
      }
    });
  });
</script>
